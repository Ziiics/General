### Turtle Creating different shapes

```
from turtle import Turtle, Screen

tim = Turtle()
tim.shape("turtle")
tim.color("cornsilk3")

color = ["salmon", "salmon1", "salmon2", "salmon3", "salmon4"]

def make_shape(num_sides):
    x = 360/num_sides
    for i in range (num_sides):
        tim.color(change_color(num_sides%5))
        tim.forward(100)
        tim.right(x)

def change_color(x):
    return color[x]

for n in range(3,100):
    make_shape(n)

screen = Screen()
screen.exitonclick()
```

### Random walk

```
import turtle as t
import random

tim = t.Turtle()
t.colormode(255)
screen = t.Screen()

direction = [0, 90, 180, 270]
tim.pensize(5)
tim.speed("fastest")

def random_color():
    r = random.randint(0, 255)
    g = random.randint(0, 255)
    b = random.randint(0, 255)
    the_color = (r, g, b)
    return the_color

def random_walk():
    turn = random.choice(direction)
    tim.forward(15)
    tim.setheading(turn)

for n in range(100):
    tim.color(random_color())
    random_walk()

screen.exitonclick()
```
