### Coffee Machine

```
import resource
import time

MENU = {
    "espresso": {
        "ingredients": {
            "water": 50,
            "coffee": 18,
        },
        "cost": 1.5,
    },
    "latte": {
        "ingredients": {
            "water": 200,
            "milk": 150,
            "coffee": 24,
        },
        "cost": 2.5,
    },
    "cappuccino": {
        "ingredients": {
            "water": 250,
            "milk": 100,
            "coffee": 24,
        },
        "cost": 3.0,
    }
}

resources = {
    "water": 300,
    "milk": 200,
    "coffee": 100,
}

money = round(0, 2)

def print_ingredients(asked):
    print_money = False
    answer = {}
    if asked == 'resources':
        answer = resources
    elif asked == 'report':
        answer = resources
        print_money = True
    else:
        answer = MENU[asked]["ingredients"]

    print(f"Water: {answer['water']}")
    if 'milk' in answer:
        print(f"Milk: {answer['milk']}")
    print(f"Coffee: {answer['coffee']}")
    if print_money:
        print(f"Money: ${money}")

def payment(item):
    global money
    quarter = 25
    dime = 10
    nickle = 5

    q_change = 0
    d_change = 0
    n_change = 0

    cost = float(MENU[item]['cost'])
    print(f"The cost is ${cost}. Please insert coins")
    num_of_quarters = float(input(f"How many quarters? "))
    num_of_dimes = float(input(f"How many dimes? "))
    num_of_nickles = float(input(f"How many nickles? "))
    total = round((0.25 * num_of_quarters) + (0.10 * num_of_dimes) + (0.05 * num_of_nickles), 2)

    change = round(total - cost, 2)
    change_for_calculation = int(change * 100)

    while change_for_calculation > 0:
        if change_for_calculation >= quarter:
            change_for_calculation -= quarter
            q_change += 1
            continue
        if change_for_calculation >= dime:
            change_for_calculation -= dime
            d_change += 1
            continue
        if change_for_calculation >= nickle:
            change_for_calculation -= nickle
            n_change += 1
            continue
        break
    give_back = f"{q_change} quarters, {d_change} dimes, and {n_change} nickles.\n"

    print(f"Currently inputted ${total:.2f}")
    print(f"Your change will be ${change:.2f}. We will give back {give_back}")

def make(the_order):
    global money
    payment(the_order)
    needed = MENU[the_order]["ingredients"]
    sufficient = True
    print("Making your order.. ")
    time.sleep(1)
    for ingredients in needed:
        if needed[ingredients] >= resources[ingredients]:
            print(f"Not enough {ingredients} to make {the_order}")
            sufficient = False
            break
        else:
            resources[ingredients] = resources[ingredients] - needed[ingredients]
    if sufficient:
        print(f"Your {the_order} is done. Here are the leftover ingredients: ")
        print_ingredients("report")


while True:
    order = input("What would you like? (espresso/latte/cappucino): ")

    # can use 'off' to turn off the machine
    if order == "off":
        print("Good bye!")
        break
    elif order == "report" or order == "resources":
        print_ingredients(order)
    else:
        make(order)
```
