### Positional vs keyword arguments

```
def the_function(param1, param2)

# to call it
the_function(param1_val, param2_val)  # positional, no definition needed
the_function(param1 = value1, param2 = value2) # keyword, can be moved around
the_function(param2 = value2, param1 = value1) # keyword, can be moved around
```

### Calculate love score

```
def calculate_love_score(person1, person2):

    combined = person1 + person2
    lower_case_combined = combined.lower()
    true = 0
    love = 0

    for letter in lower_case_combined:
        if letter is 't' or letter is 'r' or letter is 'u':
            true += 1
        elif letter is 'l' or letter is 'o' or letter is 'v':
            love += 1
        elif letter is 'e':
            love += 1
            true += 1
    response = str(true) + str(love)
    print(response)

calculate_love_score("true", "love")
```

### Caesar Code

#### Original code

```
direction = input("Type 'encode' to encrypt, type 'decode' to decrypt:\n").lower()
text = input("Type your message:\n").lower()
shift = int(input("Type the shift number:\n"))

def encrypt(message, shift_amount):
the_message = []
cypher_result = ""
for letter in message:
the_message.append(ord(letter) + shift_amount)
for index in the_message:
if index > 122:
index -= 26
cypher_result += chr(index)

    print(cypher_result)

encrypt(text, shift)
```
