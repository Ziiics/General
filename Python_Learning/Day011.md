```
import art
import random

cards = [11, 2, 3, 4, 5, 6, 7, 8, 9, 10, 10, 10, 10]
player_hand = []
computer_hand = []

def get_player_card(card) -> list:
    card_number = int(card)
    while card_number > 0:
        player_hand.append(random.choice(cards))
        card_number -= 1
    return player_hand
def get_comp_card(card) -> list:
    card_number = int(card)
    while card_number > 0:
        computer_hand.append(random.choice(cards))
        card_number -= 1
    return computer_hand

def sum_cards(deck):
    total = 0
    ace = 0
    for card in deck:
        if card == 11:
            ace += 1
        else:
            total += card
    while ace > 0:
        if total >= 11:
            total += 1
            ace -= 1
        else:
            total += 11
            ace -= 1
    return total

while True:
    print(art.logo)
    player_hand = []
    computer_hand = []

    play = input("Do you want to play a game of Blackjack? Type 'y' or 'n': ")
    if play.lower() != "y":
        print("Good bye")
        break

    get_player_card(2)
    get_comp_card(1)

    def update_situation(include_winner: bool):
        player_score = sum_cards(player_hand)
        computer_score = sum_cards(computer_hand)
        print(f"Your cards: {player_hand}, current score: {player_score}")
        if player_score > 21:
            print("Your score is over 21, thus you have lost!")
            return True
        if include_winner:
            print(f"Computer's first card: {computer_hand}, current score: {computer_score}")
            if player_score > computer_score:
                print("You win!")
            elif player_score == computer_score:
                print("You still lose!")
            else:
                print("You lose!")
        else:
            print(f"Computer's first card: {computer_hand}")
        return False

    while True:
        is_busted = update_situation(include_winner = False)
        if is_busted:
            break

        more_card = input("Type 'y' to get another card, type 'n' to pass: ")
        if more_card == "y":
            get_player_card(1)
        else:
            get_comp_card(1)
            update_situation(include_winner = True)
            break
```
