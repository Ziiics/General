#### Some rules and knowledge

#### creating class example

```
class User:

    # defined when object is created
    def __init__(self, user_id, username):
        self.id = user_id
        self.username = username

        # for optional/default attributes - listed here but not on the argument
        self.followers = 0
        self.following = 0

    # creating method inside class
    def follow(self, user):
        user.followers += 1
        self.following += 1



user_1 = User("001", "userone")
user_2 = User("002", "usertwo")

user_1.follow(user_2)
user_2.follow(user_1)
print(f"{user_1.username}, followers: {user_1.followers}")
print(user_1.followers)
print(user_1.following)
print(user_2.followers)
print(user_2.following)
```
