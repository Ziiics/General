```
def format_name(f_name, l_name):
    name = f_name.title() + " " + l_name.title()
    return name

get_f_name = input("What is your first name? ")
get_l_name = input("What is your last name? ")
print(format_name(get_f_name, get_l_name))
```

#### Leap Year

```
def is_leap_year(year):
    if year % 100 == 0:
        if year % 400 == 0:
            return True
        else:
            return False
    elif year % 4 == 0:
        return True
    else:
        return False
```

### Calculator

```
def add(n1, n2):
    return n1 + n2

def substract(n1, n2):
    return n1 - n2

def multiply(n1, n2):
    return n1 * n2

def divide(n1, n2):
    return n1 / n2

operations = {
    "+": add,
    "-": substract,
    "*": multiply,
    "/": divide
}

f_num = round(float((input("Enter your first number: "))), 1)
l_num = round(float((input("Enter your second number: "))), 1)
ops = input("Enter your operation (+,-,/,*): ")

def calculate(operator):
    if operator in operations:
        print(f"{f_num} {ops} {l_num} = {operations[operator](f_num, l_num)}")
        return operations[operator](f_num, l_num)
    else:
        return "Not the right operator"

print(calculate(ops))
```
