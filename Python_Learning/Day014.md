# Day 14 - Higher Lower Game Project

```
import game_data
import random

print("Let's start the game! \n")
data = game_data.data
random.shuffle(data)
score = 0

for i in range(0, len(data)-1):

    optA = data[i]
    optB = data[i+1]

    print("Compare A: ")
    print(f"  {optA["name"]}, a {optA["description"]}, from {optA["country"]}.")
    print("Against B: ")
    print(f"  {optB["name"]}, a {optB["description"]}, from {optB["country"]}.")
    guess = input("Who has more followers? Type 'A' or 'B': ")
    if (guess == "A" and optA["follower_count"] > optB["follower_count"]) or (guess == "B" and optB["follower_count"] > optA["follower_count"]):
        score += 1
        print("Correct")
    else:
        print(f"That's wrong. Final score: {score}")
        break
```
