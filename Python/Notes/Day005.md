# Day 5 - Python Loop

### Loops

```
# For Loop
fruits = ["Apple", "Peach", "Pear"]
for fruit in fruits:
    print(fruit)

# Loop for range
total = 0
for num in range(1,100,6):
    total += num
print(total)
```

        print("FizzBuzz")

**Password Generator**

- The longer code option because I choose to select from ordinal value instead of creating my 0won letters, numbers, or other lists

```
import random

print("Welcome to the Password Generator!\n")
letters = int(input("How many letters would you like in your password?\n"))
symbols = int(input("How many symbols would you like?\n"))
numbers = int(input("How many numbers would you like?\n"))

letter_list = []
symbol_list = []
number_list = []
passwords = []

for capital_letter in range(ord('A'), ord('Z')+1):
    letter_list.append(chr(capital_letter))
for lower_letter in range(ord('a'), ord('z')+1):
    letter_list.append(chr(lower_letter))
for number in range(1,10):
    number_list.append(str(number))
for symbols_group_1 in range(ord("!"), ord("/")+1):
    symbol_list.append(chr(symbols_group_1))
for symbols_group_2 in range(ord(":"), ord("@")+1):
    symbol_list.append(chr(symbols_group_2))
for symbols_group_3 in range(ord("["), ord("`")+1):
    symbol_list.append(chr(symbols_group_3))
for symbols_group_4 in range(ord("{"), ord("~")+1):
    symbol_list.append(chr(symbols_group_4))

random_letters = random.choices(letter_list, k=letters)
random_symbols = random.choices(symbol_list, k=symbols)
random_numbers = random.choices(number_list, k=numbers)

passwords.extend(random_letters + random_numbers + random_symbols)
random.shuffle(passwords)
the_password = ''

for thing in passwords:
    the_password += thing
print(the_password)

```
